# Validation Feedback

Validation feedback provides immediate feedback as users type, helping them understand requirements and correct errors in real-time.

## Overview

Real-time validation feedback helps users understand what's expected and correct errors before submitting the form. This improves the user experience and reduces frustration.

## Basic Validation Feedback

Provide immediate feedback as users type.

```html
<div class="form-group">
  <label for="password">Password</label>
  <input 
    type="password" 
    id="password" 
    name="password"
    aria-describedby="password-requirements password-error"
  >
  <div id="password-requirements" class="requirements">
    <p>Password must contain:</p>
    <ul>
      <li id="req-length" class="requirement">
        <span class="icon">○</span> At least 8 characters
      </li>
      <li id="req-number" class="requirement">
        <span class="icon">○</span> At least one number
      </li>
      <li id="req-uppercase" class="requirement">
        <span class="icon">○</span> At least one uppercase letter
      </li>
    </ul>
  </div>
  <span id="password-error" class="error-message" role="alert"></span>
</div>

<script>
const passwordInput = document.getElementById('password');
const reqLength = document.getElementById('req-length');
const reqNumber = document.getElementById('req-number');
const reqUppercase = document.getElementById('req-uppercase');

passwordInput.addEventListener('input', () => {
  const password = passwordInput.value;
  
  // Check length
  if (password.length >= 8) {
    reqLength.classList.add('met');
    reqLength.querySelector('.icon').textContent = '✓';
  } else {
    reqLength.classList.remove('met');
    reqLength.querySelector('.icon').textContent = '○';
  }
  
  // Check number
  if (/\d/.test(password)) {
    reqNumber.classList.add('met');
    reqNumber.querySelector('.icon').textContent = '✓';
  } else {
    reqNumber.classList.remove('met');
    reqNumber.querySelector('.icon').textContent = '○';
  }
  
  // Check uppercase
  if (/[A-Z]/.test(password)) {
    reqUppercase.classList.add('met');
    reqUppercase.querySelector('.icon').textContent = '✓';
  } else {
    reqUppercase.classList.remove('met');
    reqUppercase.querySelector('.icon').textContent = '○';
  }
});
</script>

<style>
  .requirements {
    background-color: #f5f5f5;
    padding: 0.75rem;
    border-radius: 4px;
    margin-top: 0.5rem;
  }
  
  .requirements p {
    margin: 0 0 0.5rem 0;
    font-size: 0.875rem;
  }
  
  .requirements ul {
    margin: 0;
    padding-left: 1.5rem;
    font-size: 0.875rem;
  }
  
  .requirement {
    margin-bottom: 0.25rem;
  }
  
  .requirement.met {
    color: #388e3c;
  }
  
  .requirement .icon {
    display: inline-block;
    width: 1em;
  }
</style>
```

### Key Points

- Use `aria-describedby` to associate requirements with input
- Update visual indicators as user types
- Use clear icons to show met/unmet requirements
- Provide immediate feedback

## Debounced Validation

Debounce validation to avoid excessive feedback while typing.

```html
<div class="form-group">
  <label for="email">Email Address</label>
  <input 
    type="email" 
    id="email" 
    name="email"
    aria-describedby="email-error"
  >
  <span id="email-error" class="error-message" role="alert"></span>
</div>

<script>
const emailInput = document.getElementById('email');
const emailError = document.getElementById('email-error');
let debounceTimer;

function debounce(func, wait) {
  return function(...args) {
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => func.apply(this, args), wait);
  };
}

function validateEmail() {
  const email = emailInput.value;
  const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
  
  if (email && !emailRegex.test(email)) {
    emailError.textContent = 'Please enter a valid email address (e.g., user@example.com)';
    emailInput.setAttribute('aria-invalid', 'true');
  } else {
    emailError.textContent = '';
    emailInput.setAttribute('aria-invalid', 'false');
  }
}

emailInput.addEventListener('input', debounce(validateEmail, 300));
</script>

<style>
  .error-message {
    color: #d32f2f;
    font-size: 0.875rem;
    margin-top: 0.25rem;
    display: block;
  }
</style>
```

## Password Strength Indicator

Show password strength as user types.

```html
<div class="form-group">
  <label for="password">Password</label>
  <input 
    type="password" 
    id="password" 
    name="password"
    aria-describedby="password-strength password-error"
  >
  <div id="password-strength" class="password-strength">
    <div class="strength-bar">
      <div class="strength-fill"></div>
    </div>
    <span class="strength-text"></span>
  </div>
  <span id="password-error" class="error-message" role="alert"></span>
</div>

<script>
const passwordInput = document.getElementById('password');
const strengthFill = document.querySelector('.strength-fill');
const strengthText = document.querySelector('.strength-text');

passwordInput.addEventListener('input', () => {
  const password = passwordInput.value;
  const strength = calculatePasswordStrength(password);
  
  strengthFill.style.width = `${strength.score * 25}%`;
  strengthFill.className = `strength-fill ${strength.level}`;
  strengthText.textContent = strength.label;
  
  if (password && strength.score < 2) {
    passwordInput.setAttribute('aria-invalid', 'true');
  } else {
    passwordInput.setAttribute('aria-invalid', 'false');
  }
});

function calculatePasswordStrength(password) {
  let score = 0;
  
  if (password.length >= 8) score++;
  if (password.length >= 12) score++;
  if (/[a-z]/.test(password)) score++;
  if (/[A-Z]/.test(password)) score++;
  if (/\d/.test(password)) score++;
  if (/[^a-zA-Z0-9]/.test(password)) score++;
  
  if (score <= 2) {
    return { score, level: 'weak', label: 'Weak' };
  } else if (score <= 4) {
    return { score, level: 'medium', label: 'Medium' };
  } else {
    return { score, level: 'strong', label: 'Strong' };
  }
}
</script>

<style>
  .password-strength {
    margin-top: 0.5rem;
  }
  
  .strength-bar {
    height: 4px;
    background-color: #e0e0e0;
    border-radius: 2px;
    overflow: hidden;
  }
  
  .strength-fill {
    height: 100%;
    width: 0;
    transition: width 0.3s ease, background-color 0.3s ease;
  }
  
  .strength-fill.weak {
    background-color: #d32f2f;
  }
  
  .strength-fill.medium {
    background-color: #f57c00;
  }
  
  .strength-fill.strong {
    background-color: #388e3c;
  }
  
  .strength-text {
    font-size: 0.75rem;
    margin-top: 0.25rem;
    display: block;
  }
</style>
```

## Common Mistakes

### 1. Validating Too Early

```javascript
// Bad: Validates on every keystroke without debounce
input.addEventListener('input', validate);

// Good: Debounces validation
input.addEventListener('input', debounce(validate, 300));
```

### 2. Not Using aria-invalid

```html
<!-- Bad: No aria-invalid -->
<input type="email" id="email">
<span class="error">Invalid email</span>

<!-- Good: With aria-invalid -->
<input type="email" id="email" aria-invalid="true" aria-describedby="email-error">
<span id="email-error" class="error" role="alert">Invalid email</span>
```

### 3. Not Providing Clear Feedback

```javascript
// Bad: Vague feedback
if (!isValid) {
  showError('Invalid input');
}

// Good: Specific feedback
if (!isValid) {
  showError('Please enter a valid email address (e.g., user@example.com)');
}
```

### 4. Not Updating ARIA Attributes

```javascript
// Bad: Not updating ARIA attributes
if (!isValid) {
  showError('Invalid email');
}

// Good: Updating ARIA attributes
if (!isValid) {
  input.setAttribute('aria-invalid', 'true');
  showError('Invalid email');
} else {
  input.setAttribute('aria-invalid', 'false');
  hideError();
}
```

## Best Practices

### 1. Debounce Validation

Avoid excessive feedback while typing.

```javascript
function debounce(func, wait) {
  return function(...args) {
    clearTimeout(debounceTimer);
    debounceTimer = setTimeout(() => func.apply(this, args), wait);
  };
}

input.addEventListener('input', debounce(validate, 300));
```

### 2. Validate on Blur

Validate when user leaves the field.

```javascript
input.addEventListener('blur', validate);
```

### 3. Update ARIA Attributes

Mark fields as invalid when they have errors.

```javascript
if (!isValid) {
  input.setAttribute('aria-invalid', 'true');
} else {
  input.setAttribute('aria-invalid', 'false');
}
```

### 4. Provide Clear Feedback

Tell users exactly what's wrong and how to fix it.

```javascript
if (!isValid) {
  showError('Please enter a valid email address (e.g., user@example.com)');
}
```

### 5. Use Visual Indicators

Use icons and colors to show validation status.

```html
<li class="requirement met">
  <span class="icon">✓</span> At least 8 characters
</li>
```

### 6. Don't Validate Too Early

Wait for user to finish typing before validating.

```javascript
// Validate on blur
input.addEventListener('blur', validate);

// Or debounce input validation
input.addEventListener('input', debounce(validate, 300));
```
